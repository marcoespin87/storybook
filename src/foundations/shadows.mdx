# Shadows

Defined as `--pbo-shadow-*` and mapped to `shadow-*` utilities.

  { utility: 'shadow-xs', variable: ' --pbo-shadow-xs' },
  { utility: 'shadow-sm', variable: ' --pbo-shadow-sm' },
  { utility: 'shadow-md', variable: ' --pbo-shadow-md' },
  { utility: 'shadow-lg', variable: ' --pbo-shadow-lg' },
  { utility: 'shadow-xl', variable: ' --pbo-shadow-xl' },
  { utility: 'shadow-2xl', variable: ' --pbo-shadow-2xl' }
];
